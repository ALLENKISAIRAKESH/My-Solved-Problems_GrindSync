class Solution {
    // 🔢 Mod value to avoid overflow
    private static final int MOD = 1_000_000_007;

    /*
    🔥 Recursive + Memoization Approach

    index → current position in corridor
    seats → number of 'S' (seats) in current section

    Goal:
    Divide corridor into sections such that
    each section contains EXACTLY 2 seats ('S')
    */

    private int count(int index, int seats, String corridor,
                      Map<Pair<Integer, Integer>, Integer> cache) {

        // ✅ Base case:
        // If reached end → valid only if current section has exactly 2 seats
        if (index == corridor.length()) {
            return seats == 2 ? 1 : 0;
        }

        // 🧠 Memoization check
        Pair<Integer, Integer> key = new Pair<>(index, seats);
        if (cache.containsKey(key)) {
            return cache.get(key);
        }

        int result = 0;

        // 🔥 Case 1: current section already has 2 seats
        if (seats == 2) {

            if (corridor.charAt(index) == 'S') {
                /*
                ❗ Must start new section
                because adding this 'S' would make seats > 2

                Start new section with this 'S'
                */
                result = count(index + 1, 1, corridor, cache);

            } else {
                /*
                🔀 Two choices for 'P':
                1. Close section → start new section
                2. Continue current section
                */
                result = (count(index + 1, 0, corridor, cache) +   // new section
                          count(index + 1, 2, corridor, cache)) % MOD; // continue
            }

        } else {
            /*
            🔥 Case 2: seats < 2 → keep building section
            */

            if (corridor.charAt(index) == 'S') {
                // add seat
                result = count(index + 1, seats + 1, corridor, cache);
            } else {
                // just extend with plant
                result = count(index + 1, seats, corridor, cache);
            }
        }

        // 💾 Store result in cache
        cache.put(key, result);
        return result;
    }

    public int numberOfWays(String corridor) {

        /*
        🔥 Entry point

        Start from index 0
        with 0 seats in current section
        */

        Map<Pair<Integer, Integer>, Integer> cache = new HashMap<>();

        return count(0, 0, corridor, cache);
    }
}
