class Solution {
    public boolean canPartitionGrid(int[][] grid) {

        /*
        🔥 APPROACH: Prefix Sum (Row + Column)

        From problem :contentReference[oaicite:0]{index=0}:

        Goal:
        - Make ONE horizontal OR vertical cut
        - Both parts must have equal sum

        Key idea:
        - Total sum must be EVEN
        - Try splitting:
            1. Row-wise (horizontal cut)
            2. Column-wise (vertical cut)

        Strategy:
        1. Compute total sum
        2. Try prefix sum over rows:
            if prefix == total/2 → valid cut
        3. Try prefix sum over columns:
            if prefix == total/2 → valid cut

        ⏱ Time: O(m * n)
        💾 Space: O(1)
        */

        int m = grid.length;
        int n = grid[0].length;

        long total = 0;

        // Step 1: compute total sum
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                total += grid[i][j];
            }
        }

        // total must be even
        if (total % 2 != 0) return false;

        long target = total / 2;

        // Step 2: try horizontal cuts
        long prefix = 0;

        for (int i = 0; i < m - 1; i++) { // ensure non-empty parts
            for (int j = 0; j < n; j++) {
                prefix += grid[i][j];
            }

            if (prefix == target) return true;
        }

        // Step 3: try vertical cuts
        prefix = 0;

        for (int j = 0; j < n - 1; j++) { // ensure non-empty parts
            for (int i = 0; i < m; i++) {
                prefix += grid[i][j];
            }

            if (prefix == target) return true;
        }

        return false;
    }
}
