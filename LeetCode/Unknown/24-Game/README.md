class Solution {
    public boolean judgePoint24(int[] cards) {
        java.util.List<Double> list = new java.util.ArrayList<>();
        for (int c : cards) list.add((double) c);
        return solve(list);
    }

    private boolean solve(java.util.List<Double> nums) {
        if (nums.size() == 1) return Math.abs(nums.get(0) - 24.0) < 1e-6;
        for (int i = 0; i < nums.size(); i++) {
            for (int j = 0; j < nums.size(); j++) {
                if (i == j) continue;
                java.util.List<Double> next = new java.util.ArrayList<>();
                for (int k = 0; k < nums.size(); k++) {
                    if (k != i && k != j) next.add(nums.get(k));
                }
                for (double val : compute(nums.get(i), nums.get(j))) {
                    next.add(val);
                    if (solve(next)) return true;
                    next.remove(next.size() - 1);
                }
            }
        }
        return false;
    }

    private java.util.List<Double> compute(double a, double b) {
        java.util.List<Double> res = new java.util.ArrayList<>();
        res.add(a + b);
        res.add(a - b);
        res.add(a * b);
        if (Math.abs(b) > 1e-6) res.add(a / b);
        return res;
    }
}
