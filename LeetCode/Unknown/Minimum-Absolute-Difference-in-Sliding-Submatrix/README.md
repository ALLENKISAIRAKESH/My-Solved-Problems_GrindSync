class Solution {
    public int[][] minAbsDiff(int[][] grid, int k) {

        int m = grid.length;
        int n = grid[0].length;

        int[][] ans = new int[m - k + 1][n - k + 1];

        for (int i = 0; i <= m - k; i++) {
            for (int j = 0; j <= n - k; j++) {

                // 🔥 Use set to remove duplicates
                Set<Integer> set = new HashSet<>();

                for (int x = i; x < i + k; x++) {
                    for (int y = j; y < j + k; y++) {
                        set.add(grid[x][y]);
                    }
                }

                // convert to array
                int size = set.size();

                // edge case: only 1 distinct value
                if (size == 1) {
                    ans[i][j] = 0;
                    continue;
                }

                int[] arr = new int[size];
                int idx = 0;

                for (int val : set) {
                    arr[idx++] = val;
                }

                Arrays.sort(arr);

                int minDiff = Integer.MAX_VALUE;

                for (int t = 1; t < arr.length; t++) {
                    minDiff = Math.min(minDiff, arr[t] - arr[t - 1]);
                }

                ans[i][j] = minDiff;
            }
        }

        return ans;
    }
}
