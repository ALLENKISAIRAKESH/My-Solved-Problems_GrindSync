class Solution {

    // 🔥 OPTIMIZED FINAL (better constant, less overhead, more beats)

    private static final int MOD = 1_000_000_007;

    private int modPow(long a, long b) {
        long res = 1;
        while (b > 0) {
            if ((b & 1) == 1) res = res * a % MOD;
            a = a * a % MOD;
            b >>= 1;
        }
        return (int) res;
    }

    public int xorAfterQueries(int[] nums, int[][] queries) {
        int n = nums.length;

        // 🔥 smaller threshold → better constant
        int T = 200; // tuned (instead of sqrt)

        // precompute inverses for all v (avoid repeated pow calls)
        Map<Integer, Integer> invMap = new HashMap<>();

        // group queries by k
        List<int[]>[] groups = new ArrayList[T];
        for (int i = 0; i < T; i++) groups[i] = new ArrayList<>();

        for (int[] q : queries) {
            int l = q[0], r = q[1], k = q[2], v = q[3];

            if (k < T) {
                groups[k].add(q);

                // precompute inverse once
                if (!invMap.containsKey(v)) {
                    invMap.put(v, modPow(v, MOD - 2));
                }
            } else {
                // large k → brute (few ops)
                for (int i = l; i <= r; i += k) {
                    nums[i] = (int)((long)nums[i] * v % MOD);
                }
            }
        }

        long[] dif = new long[n + T];

        for (int k = 1; k < T; k++) {
            if (groups[k].isEmpty()) continue;

            // reset only needed range (faster than full fill)
            Arrays.fill(dif, 1);

            for (int[] q : groups[k]) {
                int l = q[0], r = q[1], v = q[3];

                dif[l] = dif[l] * v % MOD;

                int R = ((r - l) / k + 1) * k + l;

                dif[R] = dif[R] * invMap.get(v) % MOD;
            }

            // prefix multiplication
            for (int i = k; i < n; i++) {
                dif[i] = dif[i] * dif[i - k] % MOD;
            }

            // apply
            for (int i = 0; i < n; i++) {
                nums[i] = (int)((long)nums[i] * dif[i] % MOD);
            }
        }

        // XOR
        int res = 0;
        for (int x : nums) res ^= x;

        return res;
    }
}
