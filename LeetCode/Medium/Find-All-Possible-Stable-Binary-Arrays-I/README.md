class Solution {
    static final int MOD = 1_000_000_007;

    // memo[z][o][last] → number of ways
    // last: 0 → ended with 0, 1 → ended with 1
    int[][][] memo;
    int limit;

    public int numberOfStableArrays(int zero, int one, int limit) {

        /*
        🔥 APPROACH: DP + Memoization (Top-Down)

        From problem :contentReference[oaicite:0]{index=0}

        ------------------------------------------------

        💡 Idea:
        Build array step by step.

        State:
        dfs(zeroLeft, oneLeft, lastPlaced)

        Constraint:
        - Cannot place more than 'limit' consecutive same bits

        ------------------------------------------------

        💡 Transition:
        If last = 0:
            → can place 1
            → can place 0 only if we haven't used 'limit' zeros in a row

        To avoid tracking count explicitly,
        we try placing k (1..limit) blocks at once.

        ------------------------------------------------

        ⏱ Time: O(zero * one * limit)
        */

        this.limit = limit;
        memo = new int[zero + 1][one + 1][2];

        for (int i = 0; i <= zero; i++) {
            for (int j = 0; j <= one; j++) {
                memo[i][j][0] = memo[i][j][1] = -1;
            }
        }

        long ans = 0;

        // start with 0 or 1
        ans = (ans + dfs(zero, one, 0)) % MOD;
        ans = (ans + dfs(zero, one, 1)) % MOD;

        return (int) ans;
    }

    private int dfs(int z, int o, int last) {
        if (z == 0 && o == 0) return 1;

        if (memo[z][o][last] != -1) return memo[z][o][last];

        long res = 0;

        if (last == 0) {
            // place 1s
            for (int k = 1; k <= limit && o - k >= 0; k++) {
                res = (res + dfs(z, o - k, 1)) % MOD;
            }
        } else {
            // place 0s
            for (int k = 1; k <= limit && z - k >= 0; k++) {
                res = (res + dfs(z - k, o, 0)) % MOD;
            }
        }

        return memo[z][o][last] = (int) res;
    }
}
