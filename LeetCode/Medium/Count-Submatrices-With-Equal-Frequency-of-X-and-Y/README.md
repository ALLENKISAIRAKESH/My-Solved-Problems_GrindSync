class Solution {
    public int numberOfSubmatrices(char[][] grid) {

        /*
        🔥 APPROACH: Prefix Sum + HashMap (2D → 1D trick)

        From problem :contentReference[oaicite:0]{index=0}:

        Conditions:
        1. equal count of 'X' and 'Y'
        2. must include grid[0][0]
        3. at least one 'X'

        ----------------------------------------------------

        💡 KEY IDEA:

        Convert:
            'X' → +1
            'Y' → -1
            '.' → 0

        Then condition:
            equal X and Y → sum == 0

        ----------------------------------------------------

        BUT:
        👉 submatrix MUST include (0,0)

        So:
        We only consider submatrices:
            from (0,0) → (i,j)

        That becomes PREFIX problem 🚀

        ----------------------------------------------------

        ALSO:
        must have at least one 'X'
        → ensure prefixX[i][j] > 0

        ----------------------------------------------------

        ⏱ Time: O(m*n)
        💾 Space: O(m*n)
        */

        int m = grid.length;
        int n = grid[0].length;

        int[][] sum = new int[m][n];
        int[][] countX = new int[m][n];

        int result = 0;

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {

                int val = 0;

                if (grid[i][j] == 'X') val = 1;
                else if (grid[i][j] == 'Y') val = -1;

                // prefix sum
                sum[i][j] = val;
                countX[i][j] = (grid[i][j] == 'X') ? 1 : 0;

                if (i > 0) {
                    sum[i][j] += sum[i - 1][j];
                    countX[i][j] += countX[i - 1][j];
                }

                if (j > 0) {
                    sum[i][j] += sum[i][j - 1];
                    countX[i][j] += countX[i][j - 1];
                }

                if (i > 0 && j > 0) {
                    sum[i][j] -= sum[i - 1][j - 1];
                    countX[i][j] -= countX[i - 1][j - 1];
                }

                // check condition
                if (sum[i][j] == 0 && countX[i][j] > 0) {
                    result++;
                }
            }
        }

        return result;
    }
}
