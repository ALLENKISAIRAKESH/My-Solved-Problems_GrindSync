class Solution {
    public int maximumAmount(int[][] coins) {
        /*
        🔥 DP with 3 states (0,1,2 neutralizations used)

        dp[i][j][k] = max coins reaching (i,j) using k neutralizations

        Transition:
        - from top or left
        - if coins[i][j] >= 0 → just add
        - if coins[i][j] < 0:
            option1 → take loss
            option2 → neutralize (if k > 0)

        Optimize space → only keep current row
        */

        int m = coins.length;
        int n = coins[0].length;

        int[][][] dp = new int[m][n][3];

        // initialize with very small values
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < 3; k++) {
                    dp[i][j][k] = Integer.MIN_VALUE / 2;
                }
            }
        }

        // start
        dp[0][0][0] = coins[0][0];
        if (coins[0][0] < 0) {
            dp[0][0][1] = 0; // neutralize
        }

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < 3; k++) {

                    if (i == 0 && j == 0) continue;

                    int val = coins[i][j];

                    // from top
                    if (i > 0) {
                        dp[i][j][k] = Math.max(dp[i][j][k],
                            dp[i-1][j][k] + val);
                    }

                    // from left
                    if (j > 0) {
                        dp[i][j][k] = Math.max(dp[i][j][k],
                            dp[i][j-1][k] + val);
                    }

                    // if negative and we can neutralize
                    if (val < 0 && k > 0) {
                        if (i > 0) {
                            dp[i][j][k] = Math.max(dp[i][j][k],
                                dp[i-1][j][k-1]);
                        }
                        if (j > 0) {
                            dp[i][j][k] = Math.max(dp[i][j][k],
                                dp[i][j-1][k-1]);
                        }
                    }
                }
            }
        }

        // answer = max over k = 0..2
        int res = Integer.MIN_VALUE;
        for (int k = 0; k < 3; k++) {
            res = Math.max(res, dp[m-1][n-1][k]);
        }

        return res;
    }
}
