import heapq

class Solution:
    def assignTasks(self, servers: list[int], tasks: list[int]) -> list[int]:
        free = [(weight, i) for i, weight in enumerate(servers)]
        heapq.heapify(free)

        busy = []
        ans = []
        time = 0

        for j in range(len(tasks)):
            time = max(time, j)

            # Release all servers that are free
            while busy and busy[0][0] <= time:
                end, weight, idx = heapq.heappop(busy)
                heapq.heappush(free, (weight, idx))

            # Wait until the earliest server becomes free
            if not free:
                time = busy[0][0]

                while busy and busy[0][0] <= time:
                    end, weight, idx = heapq.heappop(busy)
                    heapq.heappush(free, (weight, idx))

            # Assign the current task
            weight, idx = heapq.heappop(free)
            ans.append(idx)

            heapq.heappush(
                busy,
                (time + tasks[j], weight, idx)
            )

        return ans
