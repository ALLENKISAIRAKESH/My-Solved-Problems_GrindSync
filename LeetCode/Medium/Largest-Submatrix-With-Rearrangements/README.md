class Solution {
    public int largestSubmatrix(int[][] matrix) {

        /*
        🔥 APPROACH: Histogram + Sorting (Greedy)

        Idea from :contentReference[oaicite:0]{index=0}:

        Step 1: Convert matrix into "heights"
        -------------------------------------
        For each cell, count consecutive 1's vertically.

        Example:
        matrix:
        0 0 1
        1 1 1
        1 0 1

        heights:
        0 0 1
        1 1 2
        2 0 3

        -------------------------------------

        Step 2: For each row:
        - Sort heights in descending order
        - Try forming rectangles

        Why sorting works?
        Because we can rearrange columns arbitrarily.

        So largest rectangle = height * width

        -------------------------------------

        ⏱ Time: O(m * n log n)
        💾 Space: O(n)
        */

        int m = matrix.length;
        int n = matrix[0].length;

        int[][] heights = new int[m][n];

        // Step 1: Build heights
        for (int j = 0; j < n; j++) {
            heights[0][j] = matrix[0][j];
            for (int i = 1; i < m; i++) {
                if (matrix[i][j] == 1) {
                    heights[i][j] = heights[i - 1][j] + 1;
                } else {
                    heights[i][j] = 0;
                }
            }
        }

        int maxArea = 0;

        // Step 2: Process each row
        for (int i = 0; i < m; i++) {

            int[] row = heights[i].clone();

            // sort descending
            Arrays.sort(row);

            // check area
            for (int j = 0; j < n; j++) {

                int height = row[n - 1 - j]; // largest first
                int width = j + 1;

                maxArea = Math.max(maxArea, height * width);
            }
        }

        return maxArea;
    }
}
