class Solution:
    def longestPalindrome(self, s: str) -> str:
        best_l = best_r = 0
        n = len(s)
        def expand(l: int, r: int) -> None:
            nonlocal best_l, best_r
            while l >= 0 and r < n and s[l] == s[r]:
                if r - l > best_r - best_l:
                    best_l, best_r = l, r
                l -= 1
                r += 1
        for i in range(n):
            expand(i, i)
            expand(i, i + 1)

        return s[best_l : best_r + 1]
