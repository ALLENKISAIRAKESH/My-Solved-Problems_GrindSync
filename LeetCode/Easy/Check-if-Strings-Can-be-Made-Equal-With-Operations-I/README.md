class Solution {
    public boolean canBeEqual(String s1, String s2) {
        /*
        🔥 From problem :contentReference[oaicite:0]{index=0}:

        Allowed operation:
        swap indices (i, j) where j - i = 2

        👉 That means:
        - index 0 ↔ 2
        - index 1 ↔ 3

        So:
        - even indices (0,2) can permute
        - odd indices (1,3) can permute

        ✅ Condition:
        sorted(even chars of s1) == sorted(even chars of s2)
        AND
        sorted(odd chars of s1) == sorted(odd chars of s2)
        */

        char[] e1 = {s1.charAt(0), s1.charAt(2)};
        char[] o1 = {s1.charAt(1), s1.charAt(3)};

        char[] e2 = {s2.charAt(0), s2.charAt(2)};
        char[] o2 = {s2.charAt(1), s2.charAt(3)};

        java.util.Arrays.sort(e1);
        java.util.Arrays.sort(o1);
        java.util.Arrays.sort(e2);
        java.util.Arrays.sort(o2);

        return java.util.Arrays.equals(e1, e2) &&
               java.util.Arrays.equals(o1, o2);
    }
}
