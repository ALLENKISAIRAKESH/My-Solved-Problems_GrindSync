import java.util.*;

class Solution {
    public int[] finalPrices(int[] prices) {

        /*
        🔥 APPROACH: Monotonic Stack (Next Smaller or Equal Element)

        Key idea:
        - For each price[i], find the FIRST j > i such that:
              prices[j] <= prices[i]
        - That value becomes the discount

        👉 Use a MONOTONIC INCREASING stack:
        - Stack stores indices
        - While current price <= stack top → we found discount

        ⏱ Time: O(n)
        💾 Space: O(n)
        */

        int n = prices.length;
        int[] res = Arrays.copyOf(prices, n); // copy original prices

        Stack<Integer> stack = new Stack<>(); // stores indices

        for (int i = 0; i < n; i++) {

            // If current price is <= stack top → discount found
            while (!stack.isEmpty() && prices[i] <= prices[stack.peek()]) {

                int idx = stack.pop();

                // apply discount
                res[idx] = prices[idx] - prices[i];
            }

            // push current index
            stack.push(i);
        }

        return res;
    }
}
