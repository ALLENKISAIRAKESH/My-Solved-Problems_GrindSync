import java.util.*;

class Solution {
    public long countMajoritySubarrays(int[] nums, int target) {
        int n = nums.length;

        int[] prefix = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            prefix[i] = prefix[i - 1] + (nums[i - 1] == target ? 1 : -1);
        }

        int size = 2 * n + 5;
        Fenwick bit = new Fenwick(size);

        long ans = 0;

        int offset = n + 2;
        bit.add(prefix[0] + offset, 1);

        for (int i = 1; i <= n; i++) {
            ans += bit.query(prefix[i] + offset - 1);
            bit.add(prefix[i] + offset, 1);
        }

        return ans;
    }

    static class Fenwick {
        int[] tree;

        Fenwick(int n) {
            tree = new int[n + 2];
        }

        void add(int idx, int val) {
            while (idx < tree.length) {
                tree[idx] += val;
                idx += idx & -idx;
            }
        }

        int query(int idx) {
            int sum = 0;
            while (idx > 0) {
                sum += tree[idx];
                idx -= idx & -idx;
            }
            return sum;
        }
    }
}
