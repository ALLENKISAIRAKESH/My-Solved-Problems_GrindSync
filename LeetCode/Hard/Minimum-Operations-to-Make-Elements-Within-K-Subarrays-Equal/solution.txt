class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)

        window = SortedList()
        total = 0
        costs = [0] * (n - x + 1)

        for i, v in enumerate(nums):
            window.add(v)
            total += v

            if i >= x:
                old = nums[i - x]
                window.remove(old)
                total -= old

            if i >= x - 1:
                m = window[x // 2]
                left_sum = sum(window[:x // 2])
                right_sum = sum(window[x // 2 + 1:])

                costs[i - x + 1] = (
                    m * (x // 2) - left_sum
                    + right_sum - m * (x - x // 2 - 1)
                )

        prev = [0] * (n + 1)
        INF = float('inf')

        for j in range(1, k + 1):
            curr = [INF] * (n + 1)

            for i in range(1, n + 1):
                curr[i] = curr[i - 1]

                if i >= x:
                    curr[i] = min(
                        curr[i],
                        prev[i - x] + costs[i - x]
                    )

            prev = curr

        return prev[n]
