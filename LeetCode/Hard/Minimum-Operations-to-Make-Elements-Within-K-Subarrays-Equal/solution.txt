class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)
        INF = float('inf')
        left = []
        right = []
        sum_left = 0
        sum_right = 0
        def add(v):
            nonlocal sum_left, sum_right
            if not left or v <= -left[0]:
                heappush(left, -v)
                sum_left += v
            else:
                heappush(right, v)
                sum_right += v
            balance()
        def remove(v):
            nonlocal sum_left, sum_right
            if v <= -left[0]:
                sum_left -= v
                delayed_left[v] = delayed_left.get(v, 0) + 1
            else:
                sum_right -= v
                delayed_right[v] = delayed_right.get(v, 0) + 1
            prune()
            balance()
        delayed_left = {}
        delayed_right = {}
        def prune():
            while left and delayed_left.get(-left[0], 0):
                v = -heappop(left)
                delayed_left[v] -= 1
            while right and delayed_right.get(right[0], 0):
                v = heappop(right)
                delayed_right[v] -= 1
        def balance():
            nonlocal sum_left, sum_right
            prune()
            while len(left) > len(right) + 1:
                v = -heappop(left)
                sum_left -= v
                heappush(right, v)
                sum_right += v
                prune()
            while len(left) < len(right):
                v = heappop(right)
                sum_right -= v
                heappush(left, -v)
                sum_left += v
                prune()
        costs = [0] * (n - x + 1)
        for i in range(n):
            add(nums[i])
            if i >= x:
                remove(nums[i - x])
            if i >= x - 1:
                prune()
                median = -left[0]
                costs[i - x + 1] = (
                    median * len(left) - sum_left
                    + sum_right - median * len(right)
                )
        prev = [0] * (n + 1)
        for j in range(1, k + 1):
            curr = [INF] * (n + 1)
            for i in range(1, n + 1):
                curr[i] = curr[i - 1]
                if i >= x:
                    curr[i] = min(
                        curr[i],
                        prev[i - x] + costs[i - x]
                    )
            prev = curr
        return prev[n]
