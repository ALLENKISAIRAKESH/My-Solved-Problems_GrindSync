## Problem Statement

You are given an integer array nums and two integers, x and k. You can perform the following operation any number of times (including zero):


	Increase or decrease any element of nums by 1.


Return the minimum number of operations needed to have at least k non-overlapping subarrays of size exactly x in nums, where all elements within each subarray are equal.

 
Example 1:


Input: nums = [5,-2,1,3,7,3,6,4,-1], x = 3, k = 2

Output: 8

Explanation:


	Use 3 operations to add 3 to nums[1] and use 2 operations to subtract 2 from nums[3]. The resulting array is [5, 1, 1, 1, 7, 3, 6, 4, -1].
	Use 1 operation to add 1 to nums[5] and use 2 operations to subtract 2 from nums[6]. The resulting array is [5, 1, 1, 1, 7, 4, 4, 4, -1].
	Now, all elements within each subarray [1, 1, 1] (from indices 1 to 3) and [4, 4, 4] (from indices 5 to 7) are equal. Since 8 total operations were used, 8 is the output.



Example 2:


Input: nums = [9,-2,-2,-2,1,5], x = 2, k = 2

Output: 3

Explanation:


	Use 3 operations to subtract 3 from nums[4]. The resulting array is [9, -2, -2, -2, -2, 5].
	Now, all elements within each subarray [-2, -2] (from indices 1 to 2) and [-2, -2] (from indices 3 to 4) are equal. Since 3 operations were used, 3 is the output.



 
Constraints:


	2 <= nums.length <= 105
	-106 <= nums[i] <= 106
	2 <= x <= nums.length
	1 <= k <= 15
	2 <= k * x <= nums.length

class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)

        small = []  # Max heap (negative values)
        large = []  # Min heap

        delayed = defaultdict(int)

        small_size = 0
        large_size = 0

        sum_small = 0
        sum_large = 0

        def prune(heap, is_small):
            while heap:
                val = -heap[0] if is_small else heap[0]

                if delayed[val] == 0:
                    break

                delayed[val] -= 1
                heapq.heappop(heap)

        def balance():
            nonlocal small_size, large_size
            nonlocal sum_small, sum_large

            prune(small, True)
            prune(large, False)

            while small_size > large_size + 1:
                v = -heapq.heappop(small)
                sum_small -= v
                small_size -= 1

                heapq.heappush(large, v)
                sum_large += v
                large_size += 1

                prune(small, True)

            while small_size < large_size:
                v = heapq.heappop(large)
                sum_large -= v
                large_size -= 1

                heapq.heappush(small, -v)
                sum_small += v
                small_size += 1

                prune(large, False)

        def add(v):
            nonlocal small_size, large_size
            nonlocal sum_small, sum_large

            prune(small, True)

            if not small or v <= -small[0]:
                heapq.heappush(small, -v)
                small_size += 1
                sum_small += v
            else:
                heapq.heappush(large, v)
                large_size += 1
                sum_large += v

            balance()

        def remove(v):
            nonlocal small_size, large_size
            nonlocal sum_small, sum_large

            prune(small, True)

            delayed[v] += 1

            if v <= -small[0]:
                small_size -= 1
                sum_small -= v

                if v == -small[0]:
                    prune(small, True)
            else:
                large_size -= 1
                sum_large -= v

                if large and v == large[0]:
                    prune(large, False)

            balance()

        costs = [0] * (n - x + 1)

        for i in range(n):
            add(nums[i])

            if i >= x:
                remove(nums[i - x])

            if i >= x - 1:
                median = -small[0]

                costs[i - x + 1] = (
                    median * small_size - sum_small
                    + sum_large - median * large_size
                )

        # Dynamic Programming
        INF = float('inf')
        prev = [0] * (n + 1)

        for j in range(1, k + 1):
            curr = [INF] * (n + 1)

            for i in range(1, n + 1):
                curr[i] = curr[i - 1]

                if i >= x:
                    curr[i] = min(
                        curr[i],
                        prev[i - x] + costs[i - x]
                    )

            prev = curr

        return prev[n]
