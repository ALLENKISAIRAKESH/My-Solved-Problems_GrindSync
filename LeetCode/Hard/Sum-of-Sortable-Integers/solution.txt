class Solution:
    def sortableIntegers(self, nums: list[int]) -> int:
        n = len(nums)
        ans= 0
        for k in range(1, n + 1):
            if n % k:
                continue
            a = []
            ok = True
            for i in range(0, n, k):
                b = nums[i:i + k]
                c = sorted(b)
                if any(b != c[j:] + c[:j] for j in range(k)):
                    ok = False
                    break
                a.extend(c)
            if ok and a == sorted(nums):
                ans += k

        return ans
