class Fancy {

    /*
    🔥 APPROACH: Lazy Math (Affine Transformation)

    Idea from :contentReference[oaicite:0]{index=0}

    Instead of updating all elements:
        value = value * mul + add

    Maintain:
        globalMul, globalAdd

    Each element stored in "normalized form"

    ----------------------------------------

    append(val):
        store: (val - add) / mul   (using modular inverse)

    getIndex:
        actual value = stored * mul + add

    ----------------------------------------

    ⏱ All operations: O(1)
    */

    private static final int MOD = 1_000_000_007;

    private List<Long> list;
    private long mul;
    private long add;

    public Fancy() {
        list = new ArrayList<>();
        mul = 1;
        add = 0;
    }
    
    public void append(int val) {

        // normalize value before storing
        long normalized = (val - add + MOD) % MOD;

        // multiply with modular inverse of mul
        normalized = normalized * modInverse(mul) % MOD;

        list.add(normalized);
    }
    
    public void addAll(int inc) {
        add = (add + inc) % MOD;
    }
    
    public void multAll(int m) {
        mul = (mul * m) % MOD;
        add = (add * m) % MOD;
    }
    
    public int getIndex(int idx) {
        if (idx >= list.size()) return -1;

        long val = list.get(idx);

        // reconstruct actual value
        long result = (val * mul % MOD + add) % MOD;

        return (int) result;
    }

    // Fermat's Little Theorem for modular inverse
    private long modInverse(long x) {
        return modPow(x, MOD - 2);
    }

    private long modPow(long base, long exp) {
        long result = 1;
        base %= MOD;

        while (exp > 0) {
            if ((exp & 1) == 1) result = result * base % MOD;
            base = base * base % MOD;
            exp >>= 1;
        }
        return result;
    }
}
