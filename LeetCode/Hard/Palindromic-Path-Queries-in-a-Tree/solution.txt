class Solution:
    def palindromePath(self, n: int, edges: list[list[int]], s: str, queries: list[str]) -> list[bool]:
        adj = [[] for _ in range(n)]
        for u, v in edges:
            adj[u].append(v)
            adj[v].append(u)
        LOG = n.bit_length()
        tin = [0] * n
        tout = [0] * n
        depth = [0] * n
        up = [[0] * n for _ in range(LOG)]
        timer = 0
        stack = [(0, -1, 0)]
        while stack:
            u, p, state = stack.pop()
            if state == 0:
                tin[u] = timer
                timer += 1
                up[0][u] = u if p == -1 else p
                for k in range(1, LOG):
                    up[k][u] = up[k - 1][up[k - 1][u]]
                stack.append((u, p, 1))
                for v in reversed(adj[u]):
                    if v != p:
                        depth[v] = depth[u] + 1
                        stack.append((v, u, 0))
            else:
                tout[u] = timer - 1
        def lca(u, v):
            if depth[u] < depth[v]:
                u, v = v, u
            diff = depth[u] - depth[v]
            for k in range(LOG):
                if diff & (1 << k):
                    u = up[k][u]
            if u == v:
                return u
            for k in range(LOG - 1, -1, -1):
                if up[k][u] != up[k][v]:
                    u = up[k][u]
                    v = up[k][v]
            return up[0][u]
        bit = [0] * (n + 1)
        def add(i, val):
            i += 1
            while i <= n:
                bit[i] ^= val
                i += i & -i
        def prefix(i):
            res = 0
            i += 1
            while i > 0:
                res ^= bit[i]
                i -= i & -i
            return res
        def range_xor(l, r):
            if l > r:
                return 0
            return prefix(r) ^ (prefix(l - 1) if l else 0)
        chars = list(s)
        masks = [1 << (ord(c) - ord('a')) for c in chars]
        for u in range(n):
            add(tin[u], masks[u])

        ans = []

        for q in queries:
            parts = q.split()

            if parts[0] == "update":
                u = int(parts[1])
                c = parts[2]
                new_mask = 1 << (ord(c) - ord('a'))

                if masks[u] != new_mask:
                    add(tin[u], masks[u] ^ new_mask)
                    masks[u] = new_mask
                    chars[u] = c

            else:
                u, v = int(parts[1]), int(parts[2])
                w = lca(u, v)

                # XOR of root-to-node paths
                path_mask = (
                    range_xor(0, tin[u])
                    ^ range_xor(0, tin[v])
                    ^ masks[w]
                )

                # At most one character may have an odd frequency
                ans.append((path_mask & (path_mask - 1)) == 0)

        return ans
