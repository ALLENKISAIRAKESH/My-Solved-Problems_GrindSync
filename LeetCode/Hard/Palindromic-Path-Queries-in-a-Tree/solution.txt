class Solution:
    def palindromePath(self, n: int, edges: list[list[int]], s: str, queries: list[str]) -> list[bool]:
        from collections import deque

        adj = [[] for _ in range(n)]
        for u, v in edges:
            adj[u].append(v)
            adj[v].append(u)

        LOG = n.bit_length()
        tin = [0] * n
        tout = [0] * n
        depth = [0] * n
        up = [[0] * n for _ in range(LOG)]
        timer = 0

        # Doubled Euler Tour: each node appears at entry and exit.
        events = []
        stack = [(0, -1, 0)]

        while stack:
            u, p, state = stack.pop()

            if state == 0:
                tin[u] = timer
                events.append(u)
                timer += 1
                up[0][u] = u if p == -1 else p

                for k in range(1, LOG):
                    up[k][u] = up[k - 1][up[k - 1][u]]

                stack.append((u, p, 1))
                for v in reversed(adj[u]):
                    if v != p:
                        depth[v] = depth[u] + 1
                        stack.append((v, u, 0))
            else:
                tout[u] = timer
                events.append(u)
                timer += 1

        def lca(u, v):
            if depth[u] < depth[v]:
                u, v = v, u

            diff = depth[u] - depth[v]
            for k in range(LOG):
                if diff & (1 << k):
                    u = up[k][u]

            if u == v:
                return u

            for k in range(LOG - 1, -1, -1):
                if up[k][u] != up[k][v]:
                    u = up[k][u]
                    v = up[k][v]

            return up[0][u]

        bit = [0] * (2 * n + 1)

        def add(i, val):
            i += 1
            while i < len(bit):
                bit[i] ^= val
                i += i & -i

        def prefix(i):
            res = 0
            i += 1
            while i:
                res ^= bit[i]
                i -= i & -i
            return res

        chars = list(s)
        masks = [1 << (ord(c) - ord('a')) for c in chars]

        # Entry and exit events both carry the node's character.
        for i, u in enumerate(events):
            add(i, masks[u])

        def root_path(u):
            return prefix(tin[u])

        ans = []

        for q in queries:
            parts = q.split()

            if parts[0] == "update":
                u = int(parts[1])
                c = parts[2]
                new_mask = 1 << (ord(c) - ord('a'))

                if masks[u] != new_mask:
                    delta = masks[u] ^ new_mask
                    add(tin[u], delta)
                    add(tout[u], delta)
                    masks[u] = new_mask
                    chars[u] = c
            else:
                u, v = int(parts[1]), int(parts[2])
                w = lca(u, v)

                path_mask = root_path(u) ^ root_path(v) ^ masks[w]
                ans.append((path_mask & (path_mask - 1)) == 0)

        return ans
