class Solution:
    def gcdValues(self, nums: List[int], queries: List[int]) -> List[int]:
        from math import comb
        mx = max(nums)
        freq = [0] * (mx + 1)
        for num in nums:
            freq[num] += 1
        count = [0] * (mx + 1)
        for g in range(1, mx + 1):
            total = 0
            for multiple in range(g, mx + 1, g):
                total += freq[multiple]
            count[g] = total * (total - 1) // 2
        for g in range(mx, 0, -1):
            for multiple in range(g * 2, mx + 1, g):
                count[g] -= count[multiple]
        prefix = [0] * (mx + 1)
        running = 0
        for g in range(1, mx + 1):
            running += count[g]
            prefix[g] = running
        ans = []
        for q in queries:
            left, right = 1, mx
            while left < right:
                mid = (left + right) // 2
                if prefix[mid] > q:
                    right = mid
                else:
                    left = mid + 1
            ans.append(left)
        return ans
